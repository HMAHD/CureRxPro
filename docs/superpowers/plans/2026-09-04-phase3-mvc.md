spl_autoload_register(function (string $class): void {
    $prefix = 'CureRx\\';
    $base   = __DIR__ . '/';
    if (strncmp($class, $prefix, strlen($prefix)) !== 0) {
        return;
    }
    $relative = substr($class, strlen($prefix));
    $file     = $base . str_replace('\\', '/', $relative) . '.php';
    if (file_exists($file)) {
        require $file;
    }
});

<?php

namespace CureRx\Core;

class Request
{
    private string $method;
    private array  $query;
    private array  $post;
    private array  $headers;
    private ?array $jsonBody = null;

    private function __construct(
        string $method,
        array $query,
        array $post,
        array $headers
    ) {
        $this->method  = strtoupper($method);
        $this->query   = $query;
        $this->post    = $post;
        $this->headers = $headers;
    }

    public static function fromGlobals(): self
    {
        return new self(
            $_SERVER['REQUEST_METHOD'] ?? 'GET',
            $_GET,
            $_POST,
            getallheaders() ?: []
        );
    }

    public function method(): string { return $this->method; }

    public function get(string $key, mixed $default = null): mixed
    {
        return $this->query[$key] ?? $default;
    }

    public function post(string $key, mixed $default = null): mixed
    {
        return $this->post[$key] ?? $default;
    }

    public function header(string $name): ?string
    {
        $normalized = strtolower($name);
        foreach ($this->headers as $k => $v) {
            if (strtolower($k) === $normalized) {
                return $v;
            }
        }
        return null;
    }

    public function body(): array
    {
        if ($this->jsonBody === null) {
            $raw = file_get_contents('php://input');
            $this->jsonBody = json_decode($raw ?: '', true) ?? [];
        }
        return $this->jsonBody;
    }

    public function input(string $key, mixed $default = null): mixed
    {
        return $this->body()[$key] ?? $this->post[$key] ?? $this->query[$key] ?? $default;
    }
}

                $errstr,
                $errfile,
                $errline

<?php

namespace CureRx\Core;

class Validator
{
    private array $data;
    private array $errors = [];

    public function __construct(array $data)
    {
        $this->data = $data;
    }

    public function required(string $field): self
    {
        if (!isset($this->data[$field]) || $this->data[$field] === '') {
            $this->errors[$field][] = "{$field} is required";
        }
        return $this;
    }

    public function integer(string $field, ?int $min = null, ?int $max = null): self
    {
        if (!isset($this->data[$field]) || $this->data[$field] === '') {
            return $this;
        }
        if (!filter_var($this->data[$field], FILTER_VALIDATE_INT)) {
            $this->errors[$field][] = "{$field} must be an integer";
            return $this;
        }
        $val = (int)$this->data[$field];
        if ($min !== null && $val < $min) {
            $this->errors[$field][] = "{$field} must be at least {$min}";
        }
        if ($max !== null && $val > $max) {
            $this->errors[$field][] = "{$field} must be at most {$max}";
        }
        return $this;
    }

    public function decimal(string $field, ?float $min = null): self
    {
        if (!isset($this->data[$field]) || $this->data[$field] === '') {
            return $this;
        }
        if (!is_numeric($this->data[$field])) {
            $this->errors[$field][] = "{$field} must be a number";
            return $this;
        }
        if ($min !== null && (float)$this->data[$field] < $min) {
            $this->errors[$field][] = "{$field} must be at least {$min}";
        }
        return $this;
    }

    public function date(string $field): self
    {
        if (!isset($this->data[$field]) || $this->data[$field] === '') {
            return $this;
        }
        $d = \DateTime::createFromFormat('Y-m-d', $this->data[$field]);
        if (!$d || $d->format('Y-m-d') !== $this->data[$field]) {
            $this->errors[$field][] = "{$field} must be a valid date (YYYY-MM-DD)";
        }
        return $this;
    }

    public function enum(string $field, array $allowed): self
    {
        if (!isset($this->data[$field]) || $this->data[$field] === '') {
            return $this;
        }
        if (!in_array($this->data[$field], $allowed, true)) {
            $this->errors[$field][] = "{$field} must be one of: " . implode(', ', $allowed);
        }
        return $this;
    }

    public function fails(): bool
    {
        return !empty($this->errors);
    }

    public function errors(): array
    {
        return $this->errors;
    }
}

// TCP port poll — resolves true when port accepts connections, false on timeout
function waitForPort(port, host = '127.0.0.1', maxMs = 20000) {
  return new Promise((resolve) => {
    const net   = require('net');
    const start = Date.now();
    const attempt = () => {
      const sock = net.createConnection({ port, host });
      sock.on('connect', () => { sock.destroy(); resolve(true); });
      sock.on('error',   () => {
        if (Date.now() - start > maxMs) { resolve(false); }
        else { setTimeout(attempt, 300); }
      });
    };
    attempt();
  });
}

namespace CureRx\Repositories;

class PatientRepository
{
    public function __construct(private \PDO $db) {}

    public function search(string $q): array
    {
        $stmt = $this->db->prepare(
            "SELECT id, name, phone, age, gender, address, patient_code
             FROM patients
             WHERE deleted_at IS NULL
               AND (name LIKE ? OR phone LIKE ? OR patient_code LIKE ?)
             ORDER BY name ASC
             LIMIT 50"
        );
        $like = '%' . $q . '%';
        $stmt->execute([$like, $like, $like]);
        return $stmt->fetchAll(\PDO::FETCH_ASSOC);
    }

    public function find(int $id): ?array
    {
        $stmt = $this->db->prepare(
            "SELECT * FROM patients WHERE id = ? AND deleted_at IS NULL"
        );
        $stmt->execute([$id]);
        $row = $stmt->fetch(\PDO::FETCH_ASSOC);
        return $row ?: null;
    }

    public function findAll(int $limit = 100, int $offset = 0): array
    {
        $stmt = $this->db->prepare(
            "SELECT id, name, phone, age, gender, address, patient_code, created_at
             FROM patients
             WHERE deleted_at IS NULL
             ORDER BY created_at DESC
             LIMIT ? OFFSET ?"
        );
        $stmt->execute([$limit, $offset]);
        return $stmt->fetchAll(\PDO::FETCH_ASSOC);
    }

    public function create(array $data): int
    {
        $stmt = $this->db->prepare(
            "INSERT INTO patients (name, phone, age, gender, address, patient_code)
             VALUES (:name, :phone, :age, :gender, :address, :patient_code)"
        );
        $stmt->execute([
            'name'         => $data['name'],
            'phone'        => $data['phone']        ?? null,
            'age'          => $data['age']           ?? null,
            'gender'       => $data['gender']        ?? null,
            'address'      => $data['address']       ?? null,
            'patient_code' => $data['patient_code']  ?? null,
        ]);
        return (int)$this->db->lastInsertId();
    }

    public function update(int $id, array $data): bool
    {
        $stmt = $this->db->prepare(
            "UPDATE patients
             SET name = :name, phone = :phone, age = :age,
                 gender = :gender, address = :address
             WHERE id = :id AND deleted_at IS NULL"
        );
        return $stmt->execute([
            'name'    => $data['name'],
            'phone'   => $data['phone']   ?? null,
            'age'     => $data['age']     ?? null,
            'gender'  => $data['gender']  ?? null,
            'address' => $data['address'] ?? null,
            'id'      => $id,
        ]);
    }

    public function softDelete(int $id): bool
    {
        $stmt = $this->db->prepare(
            "UPDATE patients SET deleted_at = NOW() WHERE id = ? AND deleted_at IS NULL"
        );
        $stmt->execute([$id]);
        return $stmt->rowCount() > 0;
    }

<?php

namespace CureRx\Controllers;

use CureRx\Core\Request;
use CureRx\Core\Response;
use CureRx\Core\Validator;
use CureRx\Repositories\PatientRepository;

class PatientController
{
    public function __construct(private PatientRepository $patients) {}

    public function handle(Request $req): never
    {
        match ($req->method()) {
            'GET'    => $this->index($req),
            'POST'   => $this->store($req),
            'PUT'    => $this->update($req),
            'DELETE' => $this->destroy($req),
            default  => Response::error('Method not allowed', 405),
        };
    }

    private function index(Request $req): never
    {
        $q = $req->get('q', '');
        if ($q !== '') {
            Response::json(['status' => 'success', 'data' => $this->patients->search((string)$q)]);
        }
        $limit  = max(1, min(200, (int)($req->get('limit', 100))));
        $offset = max(0, (int)($req->get('offset', 0)));
        Response::json(['status' => 'success', 'data' => $this->patients->findAll($limit, $offset)]);
    }

    private function store(Request $req): never
    {
        $data = $req->body();
        $v = (new Validator($data))
            ->required('name')
            ->integer('age', 0, 150)
            ->enum('gender', ['Male', 'Female', 'Other']);
        if ($v->fails()) {
            Response::error('Validation failed', 422, $v->errors());
        }
        $id = $this->patients->create($data);
        Response::json(['status' => 'success', 'id' => $id], 201);
    }

    private function update(Request $req): never
    {
        $id   = (int)$req->get('id', 0);
        $data = $req->body();
        if ($id <= 0) {
            Response::error('Missing patient id', 400);
        }
        if ($this->patients->find($id) === null) {
            Response::error('Patient not found', 404);
        }
        $v = (new Validator($data))->required('name');
        if ($v->fails()) {
            Response::error('Validation failed', 422, $v->errors());
        }
        $this->patients->update($id, $data);
        Response::json(['status' => 'success']);
    }

    private function destroy(Request $req): never
    {
        $id = (int)$req->get('id', 0);
        if ($id <= 0) {
            Response::error('Missing patient id', 400);
        }
        if (!$this->patients->softDelete($id)) {
            Response::error('Patient not found', 404);
        }
        Response::json(['status' => 'success']);
    }
}

namespace CureRx\Services;

class QueueService
{
    public function __construct(private \PDO $db) {}

    public function assignToken(int $doctorId): int
    {
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare(
                "SELECT COALESCE(MAX(token_number), 0) + 1
                 FROM waiting_queue
                 WHERE doctor_id = ? AND DATE(created_at) = CURDATE()
                 FOR UPDATE"
            );
            $stmt->execute([$doctorId]);
            $token = (int)$stmt->fetchColumn();
            $this->db->commit();
            return $token;
        } catch (\Throwable $e) {
            $this->db->rollBack();
            throw $e;
        }
    }

    public function addToQueue(int $doctorId, int $patientId, string $patientName): array
    {
        $token = $this->assignToken($doctorId);
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare(
                "INSERT INTO waiting_queue
                    (doctor_id, patient_id, patient_name, token_number, status)
                 VALUES (?, ?, ?, ?, 'Waiting')"
            );
            $stmt->execute([$doctorId, $patientId, $patientName, $token]);
            $id  = (int)$this->db->lastInsertId();
            $this->db->commit();
            return ['id' => $id, 'token_number' => $token];
        } catch (\Throwable $e) {
            $this->db->rollBack();
            throw $e;
        }
    }
}

        ]);

            ->required('doctor_id')
            ->required('patient_id')
            ->required('patient_name')
            ->integer('doctor_id', 1)
            ->integer('patient_id', 1);
        if ($v->fails()) {
            Response::error('Validation failed', 422, $v->errors());

        $result = $this->service->addToQueue(
            (int)$data['doctor_id'],
            (int)$data['patient_id'],
            (string)$data['patient_name']
        );
        Response::json(['status' => 'success', 'data' => $result], 201);
    }

        $id   = (int)$req->get('id', 0);
        $data = $req->body();

            ->required('status')
            ->enum('status', ['Waiting', 'Serving', 'Done']);

        }
        if (!$this->queue->updateStatus($id, $data['status'])) {
            Response::error('Queue entry not found', 404);
        }
        Response::json(['status' => 'success']);

namespace CureRx\Repositories;

class TemplateRepository
{
    public function __construct(private \PDO $db) {}

    public function findAll(): array
    {
        $templates = $this->db->query(
            "SELECT t.id, t.name, t.diagnosis,
                    ti.id as item_id, ti.medicine_name, ti.strength,
                    ti.dosage, ti.instruction, ti.duration, ti.quantity
             FROM templates t
             LEFT JOIN template_items ti ON ti.template_id = t.id
             ORDER BY t.id ASC"
        )->fetchAll(\PDO::FETCH_ASSOC);

        return $this->groupItems($templates);
    }

    private function groupItems(array $rows): array
    {
        $result = [];
        foreach ($rows as $row) {
            $tid = $row['id'];
            if (!isset($result[$tid])) {
                $result[$tid] = [
                    'id'        => $row['id'],
                    'name'      => $row['name'],
                    'diagnosis' => $row['diagnosis'],
                    'items'     => [],
                ];
            }
            if ($row['item_id'] !== null) {
                $result[$tid]['items'][] = [
                    'id'            => $row['item_id'],
                    'medicine_name' => $row['medicine_name'],
                    'strength'      => $row['strength'],
                    'dosage'        => $row['dosage'],
                    'instruction'   => $row['instruction'],
                    'duration'      => $row['duration'],
                    'quantity'      => $row['quantity'],
                ];
            }
        }
        return array_values($result);
    }

    public function create(array $data): int
    {
        $this->db->beginTransaction();
        try {
            $stmt = $this->db->prepare(
                "INSERT INTO templates (name, diagnosis) VALUES (:name, :diagnosis)"
            );
            $stmt->execute(['name' => $data['name'], 'diagnosis' => $data['diagnosis'] ?? null]);
            $tid = (int)$this->db->lastInsertId();

            foreach ($data['items'] ?? [] as $item) {
                $this->db->prepare(
                    "INSERT INTO template_items
                        (template_id, medicine_name, strength, dosage, instruction, duration, quantity)
                     VALUES (?, ?, ?, ?, ?, ?, ?)"
                )->execute([
                    $tid,
                    $item['medicine_name'] ?? '',
                    $item['strength']      ?? null,
                    $item['dosage']        ?? null,
                    $item['instruction']   ?? null,
                    $item['duration']      ?? null,
                    $item['quantity']      ?? null,
                ]);
            }
            $this->db->commit();
            return $tid;
        } catch (\Throwable $e) {
            $this->db->rollBack();
            throw $e;
        }
    }

    public function delete(int $id): bool
    {
        $stmt = $this->db->prepare("DELETE FROM templates WHERE id = ?");
        $stmt->execute([$id]);
        return $stmt->rowCount() > 0;
    }
}

                    pi.instruction, pi.duration, pi.quantity, pi.unit_price, pi.item_total
             FROM prescriptions p
             LEFT JOIN prescription_items pi ON pi.prescription_id = p.id
             WHERE p.id = ? AND p.deleted_at IS NULL"

    {
        $stmt = $this->db->prepare(
            "UPDATE prescriptions SET deleted_at = NOW() WHERE id = ? AND deleted_at IS NULL"
        );
        $stmt->execute([$id]);
        return $stmt->rowCount() > 0;

    {
        match ($req->method()) {
            'GET'    => $this->index($req),
            'POST'   => $this->store($req),
            'DELETE' => $this->destroy($req),
            default  => Response::error('Method not allowed', 405),
        };
    }

    private function index(Request $req): never
    {
        $patientId = (int)$req->get('patient_id', 0);
        if ($patientId > 0) {
            Response::json(['status' => 'success', 'data' => $this->prescriptions->findAllForPatient($patientId)]);
        }
        $id = (int)$req->get('id', 0);
        if ($id > 0) {
            $rx = $this->prescriptions->find($id);
            if (!$rx) {
                Response::error('Prescription not found', 404);
            }
            Response::json(['status' => 'success', 'data' => $rx]);
        }
        Response::error('patient_id or id required', 400);
    }

    private function store(Request $req): never
    {
        $data = $req->body();
        $v = (new Validator($data))
            ->required('patient_id')
            ->required('patient_name')
            ->integer('patient_id', 1)
            ->decimal('consultation_fee', 0);
        if ($v->fails()) {
            Response::error('Validation failed', 422, $v->errors());
        }
        $id = $this->service->create($data);
        Response::json(['status' => 'success', 'id' => $id], 201);
    }

    private function destroy(Request $req): never
    {
        $id = (int)$req->get('id', 0);
        if (!$this->prescriptions->softDelete($id)) {
            Response::error('Prescription not found', 404);
        }
        Response::json(['status' => 'success']);
    }
}

<?php

namespace CureRx\Controllers;

use CureRx\Core\Request;
use CureRx\Core\Response;
use CureRx\Repositories\SettingsRepository;

class SettingsController
{
    public function __construct(private SettingsRepository $settings) {}

    public function handle(Request $req): never
    {
        match ($req->method()) {
            'GET'  => Response::json(['status' => 'success', 'data' => $this->settings->get()]),
            'POST' => $this->update($req),
            default => Response::error('Method not allowed', 405),
        };
    }

    private function update(Request $req): never
    {
        $data = $req->body();

        // If license_key present, validate before saving
        if (!empty($data['license_key'])) {
            $secretFile = (getenv('USER_DATA_PATH') ?: __DIR__ . '/../../') . '/api_secret.key';
            $secret     = file_exists($secretFile) ? trim(file_get_contents($secretFile)) : '';
            $result     = $this->validateLicenseKey($data['license_key'], $secret);
            if (!$result['valid']) {
                Response::error('Invalid or expired license key', 422);
            }
        }

        $this->settings->update($data);
        Response::json(['status' => 'success']);
    }

    private function validateLicenseKey(string $key, string $secret): array
    {
        if (!str_starts_with($key, 'CRX-')) {
            return ['valid' => false];
        }
        $parts = explode('-', $key, 3);
        if (count($parts) !== 3) {
            return ['valid' => false];
        }
        [, $payload, $providedHmac] = $parts;
        $signingKey   = hash('sha256', $secret, true);
        $expectedHmac = hash_hmac('sha256', $payload, $signingKey);
        if (!hash_equals($expectedHmac, $providedHmac)) {
            return ['valid' => false];
        }
        $decoded = base64_decode(strtr($payload, '-_', '+/'));
        if ($decoded === false) {
            return ['valid' => false];
        }
        [$plan, $expiry] = explode('|', $decoded, 2) + ['', '0'];
        if ((int)$expiry !== 0 && (int)$expiry < time()) {
            return ['valid' => false];
        }
        return ['valid' => in_array($plan, ['starter', 'pro', 'clinic'], true)];
    }
}
